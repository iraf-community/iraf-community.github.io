Hi!
I have a script that I wrote several years ago. The script took a data
cube and made it two-dimensional: basically I took each plane of the
cube and laied it one next to another which obviously formed a long stripe.

Then we upgraded to Iraf 2.11 and I found that it yells at me saying that
&quot;imdebug&quot; amd &quot;mkimage&quot; tasks (that I am using in the script) are not
found. Could someone, please, explain what happened to those tasks in 2.11
and it they disappeared to the blue, which ones I should use instead.

I am also attaching my script. Thanks in advance,

	Sergei
	
procedure layout (PDSim,Output)

# This script takes a cubic image and makes a 2D image out of it.
# It copies each plane if the cube and adds it at the
# tail of the previous.

string PDSim			{prompt=&quot;PDS image to crash&quot;}
string Output			{prompt=&quot;Output image&quot;}

begin
	int i
	int X,Y,Z,ii
	string Type
	imdebug
	
	imgets(PDSim,&quot;i_naxis1&quot;)
	X=int(imgets.value)
	imgets(PDSim,&quot;i_naxis2&quot;)
	Y=int(imgets.value)
	imgets(PDSim,&quot;i_naxis3&quot;)
	Z=int(imgets.value)
	print(&quot;X=&quot;,X,&quot;Y=&quot;,Y,&quot;Z=&quot;,Z)
	imgets(PDSim,&quot;i_pixtype&quot;)
	ii=int(imgets.value)
	switch(ii) {
            case 3 :     Type=&quot;short&quot;
	    case 4 :     Type=&quot;int&quot;
	    case 6 :     Type=&quot;real&quot;
	   default :   error(1,&quot;Unknown type of pixels&quot;)
        }	
	print(&quot;Making target image &quot;,X,&quot;x &quot;,Z*Y)
	mkimage(Output,X,Z*Y,Type,&quot;0&quot;,PDSim)
	print(&quot;Forming plane of scans &quot;,Output,&quot; from &quot;,PDSim);
	imcopy(PDSim//&quot;[*,*,1]&quot;,Output//&quot;[1:&quot;//X//&quot;,1:&quot;//Y//&quot;]&quot;)
	for(i=2;i&lt;=Z;i+=1) {
	    imcopy(PDSim//&quot;[*,*,&quot;//i//&quot;]&quot;,
		   Output//&quot;[1:&quot;//X//&quot;,&quot;//(i-1)*35//&quot;:&quot;//i*Y//&quot;]&quot;)
	}
end
