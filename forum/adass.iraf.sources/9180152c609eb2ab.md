# SPECTYPE -- Compare spectrum against a library of spectra.
# The library parameter is a file of spectrum names in some desired order.
#
# This task uses SPECPLOT to plot the spectrum between two library spectra
# and then uses a cursor read to get the next command.  So users must first
# type &#x27;q&#x27; to quit SPECPLOT and then type one of the keys of this task.
#
# n - go to next library spectrum
# p - go to previous library spectrum
# r - redraw with current library spectrum
# x - exit
#
# To install this task:
#
# 1. Copy script to your IRAF home directory.
# 2. In your loginuser.cl or login.cl enter before the final keep:
#        task spectype=home$spectype.cl
#
# Note that the library is a text file of single comparison spectra.  If
# the spectra are in a single multispec file or multiextension fits you would
# use a syntax like abc.ms[*,3] or mef.fits[3] for each spectrum.

procedure spectype (spectrum)

string	spectrum		{prompt=&quot;Spectrum to type&quot;}
string	library			{prompt=&quot;File of comparison spectra&quot;}
pset	specplot		{prompt=&quot;Specplot parameters&quot;}

struct	*fd

begin
	string	spec, spec1, spec2, temp, key
	real	xcur, ycur
	int	wcs
	struct	cmd
	bool	plot

	# Check for SPECPLOT.
	if (!deftask (&quot;specplot&quot;))
	    error (5, &quot;First load package with SPECPLOT&quot;)

	# Query parameters.
	spec = spectrum

	# Check parameters.
	if (!imaccess (spec))
	    error (1, &quot;Spectrum not available&quot;)
	if (!access (library))
	    error (2, &quot;Library not available&quot;)

	fd = library
	if (fscan (fd, spec1) == EOF)
	    error (3, &quot;Library is empty&quot;)
	if (fscan (fd, spec2) == EOF)
	    error (4, &quot;Library contains only one spectrum&quot;)

	key = &quot;r&quot;
	while (key != &quot;x&quot;) {
	    if (key == &quot;n&quot;) {
		temp = spec1
		if (fscan (fd, spec1) == EOF) {
		    printf (&quot;At end of library\n&quot;)
		    beep
		    plot = no
		} else {
		    spec2 = temp
		    plot = yes
		}
	    } else if (key == &quot;p&quot;) {
		fd = library
		temp = spec1
		i = fscan (fd, spec1)
		if (spec1 == temp) {
		    printf (&quot;At beginning of library\n&quot;)
		    beep
		    plot = no
		} else {
		    while (fscan (fd, spec2) != EOF) {
			if (spec2 == temp)
			    break
			spec1 = spec2
		    }
		    plot = yes
		}
	    } else if (key == &quot;r&quot;) {
		plot = yes
	    } else {
		printf (&quot;Unrecognized key: %s\n&quot;, key)
		beep
	    }

	    if (plot) {
		specplot (spec2//&quot;,&quot;//spec//&quot;,&quot;//spec1)
		plot = no
	    }
	    printf (&quot;Command: (x=exit, n=next, p=previos, r=redraw)\n&quot;)
	    if (fscan (gcur, xcur, ycur, wcs, key, cmd) == EOF)
		break
	}
	fd = &quot;&quot;
end
