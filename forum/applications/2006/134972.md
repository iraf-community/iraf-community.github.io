procedure   offset(num,inlist,outlist)<br>
<br>
#######################################################################<br>
#     this procedure reads the header, searches for a certain line,                  <br>
#     i.g. "CUMOFFSETX" and writes its value in a new file.            <br>
#     This file containes than the offset in x and y direction in     <br>
#                                                                     <br>
#     start procedure with                                            <br>
#     task offset ="/home/evol/*/iraf/offset.cl"                      <br>
#######################################################################<br>
<br>
string      inlist     {prompt="List of images, e.g. nfdsz.slst"}<br>
string      outlist    {prompt="Name of offset file"}<br>
int         num        {prompt="Nuber of images"}<br>
<br>
struct     *imgstruc       # contains file list<br>
<br>
begin<br>
 <br>
           string         lfl                 # list containing images<br>
           string         nm               # name off offset file<br>
           string         cim              # current image<br>
           real           xoff               # XOFFSET<br>
           real           yoff               # YOFFSET<br>
           int            number           # number of image<br>
           string         dum              # dummy variable<br>
           int            i                        # running number<br>
           real           xsh,ysh          # shifts<br>
           string         gotit            # for fscan<br>
<br>
# prompt for input values<br>
<br>
         nm       = outlist//".off"<br>
         lfl      = inlist//".slst"<br>
         number   = num<br>
<br>
         delete (nm)<br>
<br>
         imgstruc = lfl<br>
<br>
         for (i=1; i&lt;= num; i+=1)      {<br>
         gotit = fscan(imgstruc,cim)<br>
         print (" Processing  "//cim)<br>
         imhead (cim) | match "CUMOFFSETX  "  | scan (dum,dum,dum,dum,dum,xoff)<br>
         imhead (cim) | match "CUMOFFSETY  "  | scan (dum,dum,dum,dum,dum,yoff)<br>
         xsh = -xoff<br>
         ysh = -yoff<br>
         printf("%10.5f   %10.5f\n",xsh,ysh, &gt;&gt; nm)<br>
         }<br>
<br>
        end
